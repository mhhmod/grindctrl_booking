# Try-On UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the merchant able to see everything they configure, make the icon gradient choosable without colour theory, and make the shopper's generated image the hero with full-screen zoom and reusable photos.

**Architecture:** Merchant work lands in the settings form shared by the Shopify embedded admin and the GrindCTRL dashboard, so both surfaces change once. Shopper work lands in the embed panel, with the one piece that cannot live in an iframe (true full-screen) handled by a new postMessage type on the existing parent bridge. Every new string is bilingual, every new surface reads merchant theme tokens, and the saved-photo cache is device-only.

**Tech Stack:** Next.js 15 App Router, TypeScript, Tailwind v4, shadcn/ui, Supabase (settings only), vitest, Shopify theme app extension (vanilla JS).

```ts
// apps/web-next/lib/try-on/gradient.ts
/* One brand colour to a two-stop gradient.

   The settings form used to ask a merchant for a "start" and an "end"
   colour. Choosing two colours that work together is a designer's job, so
   this derives the partner stop instead: same hue family, rotated slightly
   and lifted in lightness, with one intensity control for how far apart the
   two stops sit.

   Pure and dependency-free so it can be unit tested and shared by the form,
   the preview and the storefront without pulling in a colour library. */

export type GradientStops = { from: string; to: string };

export const DEFAULT_GRADIENT_INTENSITY = 55;

/* Used when the stored colour cannot be parsed. Neutral rather than branded:
   a wrong-but-visible control is easier to notice and fix than a black hole. */
const FALLBACK_FROM = '#3a3a3a';
const FALLBACK_TO = '#6b6b6b';

function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

function parseHex(value: string): { r: number; g: number; b: number } | null {
  const match = /^#?([0-9a-f]{6})$/i.exec(String(value ?? '').trim());
  if (!match) return null;
  const int = Number.parseInt(match[1], 16);
  return { r: (int >> 16) & 255, g: (int >> 8) & 255, b: int & 255 };
}

function toHex(r: number, g: number, b: number): string {
  const part = (n: number) =>
    Math.round(clamp(n, 0, 255)).toString(16).padStart(2, '0');
  return `#${part(r)}${part(g)}${part(b)}`;
}

function rgbToHsl(r: number, g: number, b: number) {
  const rn = r / 255;
  const gn = g / 255;
  const bn = b / 255;
  const max = Math.max(rn, gn, bn);
  const min = Math.min(rn, gn, bn);
  const l = (max + min) / 2;
  const d = max - min;

  if (d === 0) return { h: 0, s: 0, l };

  const s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
  let h: number;
  if (max === rn) h = ((gn - bn) / d + (gn < bn ? 6 : 0)) * 60;
  else if (max === gn) h = ((bn - rn) / d + 2) * 60;
  else h = ((rn - gn) / d + 4) * 60;

  return { h, s, l };
}

function hslToHex(h: number, s: number, l: number): string {
  const c = (1 - Math.abs(2 * l - 1)) * s;
  const hp = (((h % 360) + 360) % 360) / 60;
  const x = c * (1 - Math.abs((hp % 2) - 1));
  const [r1, g1, b1] =
    hp < 1 ? [c, x, 0]
    : hp < 2 ? [x, c, 0]
    : hp < 3 ? [0, c, x]
    : hp < 4 ? [0, x, c]
    : hp < 5 ? [x, 0, c]
    : [c, 0, x];
  const m = l - c / 2;
  return toHex((r1 + m) * 255, (g1 + m) * 255, (b1 + m) * 255);
}

export function normalizeHex(value: string): string | null {
  const rgb = parseHex(value);
  return rgb ? toHex(rgb.r, rgb.g, rgb.b) : null;
}

export function deriveGradient(baseHex: string, intensity: number): GradientStops {
  const rgb = parseHex(baseHex);
  if (!rgb) return { from: FALLBACK_FROM, to: FALLBACK_TO };

  const amount =
    (Number.isFinite(intensity) ? clamp(intensity, 0, 100) : DEFAULT_GRADIENT_INTENSITY) / 100;

  const { h, s, l } = rgbToHsl(rgb.r, rgb.g, rgb.b);

  /* Lift toward white, but only across the room that is left. A pale brand
     colour would otherwise clip at #ffffff and produce two identical stops,
     which reads as "the gradient is broken". */
  const headroom = 1 - l;
  const lifted = l + headroom * (0.18 + amount * 0.55);

  /* A dark colour has no headroom problem but does go muddy if the partner
     stop keeps full saturation, so ease saturation off as we lighten. */
  const softened = s * (1 - amount * 0.18);

  return {
    from: toHex(rgb.r, rgb.g, rgb.b),
    to: hslToHex(h + amount * 22, clamp(softened, 0, 1), clamp(lifted, 0, 0.97)),
  };
}
```

```tsx
// apps/web-next/components/try-on/gradient-field.tsx
'use client';

import * as React from 'react';
import { useId, useState } from 'react';
import { Label } from '@/components/ui/label';
import {
  DEFAULT_GRADIENT_INTENSITY,
  deriveGradient,
  normalizeHex,
} from '@/lib/try-on/gradient';
import type { SettingsFormCopy } from '@/lib/try-on/settings-copy';

/* Three tiers over the same two stored values: presets for the merchant who
   wants it to look right, a single brand colour for the one who has a brand,
   and the original two pickers for anyone who already tuned a pair. Nothing
   here changes what is stored, so the widget is unaffected. */

export const GRADIENT_PRESETS: ReadonlyArray<{ from: string; to: string }> = [
  { from: '#ff9a3d', to: '#ffd76e' },
  { from: '#6ab7ff', to: '#a7d9ff' },
  { from: '#5cc98d', to: '#b6f0cd' },
  { from: '#c084fc', to: '#f0abfc' },
  { from: '#f472b6', to: '#fda4af' },
  { from: '#3a3a3a', to: '#6b6b6b' },
  { from: '#eb7805', to: '#ffb25c' },
  { from: '#1f7a4d', to: '#5cc98d' },
];

export function GradientField({
  from,
  to,
  copy,
  onChange,
}: {
  from: string;
  to: string;
  copy: SettingsFormCopy;
  onChange: (next: { from: string; to: string }) => void;
}) {
  const id = useId();
  const [mode, setMode] = useState<'presets' | 'mine' | 'advanced'>('presets');
  const [intensity, setIntensity] = useState(DEFAULT_GRADIENT_INTENSITY);

  const activePreset = GRADIENT_PRESETS.findIndex(
    (p) => p.from.toLowerCase() === from.toLowerCase() && p.to.toLowerCase() === to.toLowerCase(),
  );

  function applyDerived(base: string, strength: number) {
    const normalized = normalizeHex(base);
    if (!normalized) return;
    onChange(deriveGradient(normalized, strength));
  }

  return (
    <div className="grid gap-3">
      <Label htmlFor={id}>{copy.gradientPresets}</Label>

      <div id={id} className="flex flex-wrap gap-2">
        {GRADIENT_PRESETS.map((preset, index) => (
          <button
            key={preset.from + preset.to}
            type="button"
            aria-label={`${preset.from} ${preset.to}`}
            aria-pressed={index === activePreset}
            onClick={() => {
              setMode('presets');
              onChange(preset);
            }}
            className="size-10 rounded-xl border border-input transition-transform hover:-translate-y-0.5 focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50 aria-pressed:ring-2 aria-pressed:ring-foreground"
            style={{ background: `linear-gradient(135deg, ${preset.from}, ${preset.to})` }}
          />
        ))}
      </div>

      <div className="flex flex-wrap gap-3 text-sm">
        <button
          type="button"
          onClick={() => setMode(mode === 'mine' ? 'presets' : 'mine')}
          className="rounded-full border border-input px-3 py-1 transition-colors hover:bg-muted"
        >
          {copy.gradientUseMyColour}
        </button>
        <button
          type="button"
          onClick={() => setMode(mode === 'advanced' ? 'presets' : 'advanced')}
          className="rounded-full px-3 py-1 text-muted-foreground underline-offset-4 hover:underline"
        >
          {copy.gradientAdvanced}
        </button>
      </div>

      {mode === 'mine' && (
        <div className="grid gap-3 rounded-xl border border-input p-3">
          <div className="grid gap-2">
            <Label htmlFor={`${id}-base`}>{copy.gradientBaseColour}</Label>
            <input
              id={`${id}-base`}
              type="color"
              value={normalizeHex(from) ?? '#ff9a3d'}
              onChange={(e) => applyDerived(e.target.value, intensity)}
              className="h-11 w-full cursor-pointer rounded-lg border border-input bg-background"
            />
          </div>
          <div className="grid gap-2">
            <Label htmlFor={`${id}-intensity`}>{copy.gradientIntensity}</Label>
            <input
              id={`${id}-intensity`}
              type="range"
              min={0}
              max={100}
              value={intensity}
              onChange={(e) => {
                const next = Number(e.target.value);
                setIntensity(next);
                applyDerived(from, next);
              }}
              className="h-11 w-full accent-foreground"
            />
          </div>
        </div>
      )}

      {mode === 'advanced' && (
        <div className="grid gap-3 rounded-xl border border-input p-3 sm:grid-cols-2">
          <div className="grid gap-2">
            <Label htmlFor={`${id}-from`}>{copy.iconGradientStart}</Label>
            <input
              id={`${id}-from`}
              type="color"
              value={normalizeHex(from) ?? '#ff9a3d'}
              onChange={(e) => onChange({ from: e.target.value, to })}
              className="h-11 w-full cursor-pointer rounded-lg border border-input bg-background"
            />
          </div>
          <div className="grid gap-2">
            <Label htmlFor={`${id}-to`}>{copy.iconGradientEnd}</Label>
            <input
              id={`${id}-to`}
              type="color"
              value={normalizeHex(to) ?? '#ffd76e'}
              onChange={(e) => onChange({ from, to: e.target.value })}
              className="h-11 w-full cursor-pointer rounded-lg border border-input bg-background"
            />
          </div>
        </div>
      )}
    </div>
  );
}
```

### Task 10: The viewer dialog

**Files:**
- Create: `apps/web-next/components/try-on/image-viewer.tsx`

- [ ] **Step 1: Write the component**

```tsx
// apps/web-next/components/try-on/image-viewer.tsx
'use client';

import * as React from 'react';
import { useCallback, useEffect, useRef, useState } from 'react';

/* Full-screen result viewer.

   Built on <dialog> so focus trapping, Escape-to-close and inertness of the
   page behind come from the platform instead of being hand-rolled. Zoom is
   a transform on the image, so panning stays on the compositor and does not
   repaint. */

const MIN_SCALE = 1;
const MAX_SCALE = 4;

export function ImageViewer({
  src,
  alt,
  open,
  onClose,
  labels,
}: {
  src: string;
  alt: string;
  open: boolean;
  onClose: () => void;
  labels: { close: string; zoomIn: string; zoomOut: string };
}) {
  const ref = useRef<HTMLDialogElement>(null);
  const [scale, setScale] = useState(1);
  const [offset, setOffset] = useState({ x: 0, y: 0 });
  const drag = useRef<{ x: number; y: number } | null>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    if (open && !el.open) {
      el.showModal();
      setScale(1);
      setOffset({ x: 0, y: 0 });
    }
    if (!open && el.open) el.close();
  }, [open]);

  const zoomBy = useCallback((delta: number) => {
    setScale((s) => Math.min(MAX_SCALE, Math.max(MIN_SCALE, s + delta)));
  }, []);

  return (
    <dialog
      ref={ref}
      onClose={onClose}
      onCancel={onClose}
      className="m-0 h-dvh max-h-none w-screen max-w-none bg-black/95 p-0 backdrop:bg-black/80"
    >
      <div className="relative flex h-full w-full items-center justify-center overflow-hidden">
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img
          src={src}
          alt={alt}
          draggable={false}
          onWheel={(e) => zoomBy(e.deltaY > 0 ? -0.2 : 0.2)}
          onPointerDown={(e) => {
            drag.current = { x: e.clientX - offset.x, y: e.clientY - offset.y };
            e.currentTarget.setPointerCapture(e.pointerId);
          }}
          onPointerMove={(e) => {
            if (!drag.current) return;
            setOffset({ x: e.clientX - drag.current.x, y: e.clientY - drag.current.y });
          }}
          onPointerUp={() => {
            drag.current = null;
          }}
          className="max-h-full max-w-full touch-none select-none object-contain"
          style={{
            transform: `translate3d(${offset.x}px, ${offset.y}px, 0) scale(${scale})`,
            cursor: scale > 1 ? 'grab' : 'default',
          }}
        />

        <div className="absolute inset-x-0 bottom-0 flex justify-center gap-2 p-4">
          <button
            type="button"
            aria-label={labels.zoomOut}
            onClick={() => zoomBy(-0.4)}
            className="size-11 rounded-full bg-white/15 text-white backdrop-blur"
          >
            &minus;
          </button>
          <button
            type="button"
            aria-label={labels.zoomIn}
            onClick={() => zoomBy(0.4)}
            className="size-11 rounded-full bg-white/15 text-white backdrop-blur"
          >
            +
          </button>
        </div>

        <button
          type="button"
          aria-label={labels.close}
          onClick={onClose}
          className="absolute end-3 top-3 size-11 rounded-full bg-white/15 text-white backdrop-blur"
        >
          &times;
        </button>
      </div>
    </dialog>
  );
}
```

```ts
// apps/web-next/lib/try-on/photo-cache.ts
/* Device-only cache of photos the shopper has already used.

   Deliberately never leaves the device. The app stores no customer images
   server-side, and keeping that true is worth more than cross-device
   convenience: server storage would mean consent, retention, deletion and
   disclosure obligations for photos of people.

   Browsers partition third-party iframe storage per top-level site, so this
   cache is naturally scoped to one merchant's store. That is the desirable
   behaviour, not a limitation to work around. */

export type SavedPhoto = { id: string; dataUrl: string; savedAt: number };

export const MAX_SAVED_PHOTOS = 6;

const DB_NAME = 'grindctrl-tryon';
const STORE = 'photos';

export function isPhotoCacheAvailable(): boolean {
  try {
    return typeof indexedDB !== 'undefined' && indexedDB !== null;
  } catch {
    // Accessing indexedDB can throw outright under some privacy settings.
    return false;
  }
}

export function trimToLimit(photos: SavedPhoto[]): SavedPhoto[] {
  return [...photos].sort((a, b) => b.savedAt - a.savedAt).slice(0, MAX_SAVED_PHOTOS);
}

function openDb(): Promise<IDBDatabase | null> {
  if (!isPhotoCacheAvailable()) return Promise.resolve(null);
  return new Promise((resolve) => {
    const request = indexedDB.open(DB_NAME, 1);
    request.onupgradeneeded = () => {
      if (!request.result.objectStoreNames.contains(STORE)) {
        request.result.createObjectStore(STORE, { keyPath: 'id' });
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => resolve(null);
  });
}

export async function listSavedPhotos(): Promise<SavedPhoto[]> {
  const db = await openDb();
  if (!db) return [];
  return new Promise((resolve) => {
    const request = db.transaction(STORE, 'readonly').objectStore(STORE).getAll();
    request.onsuccess = () => resolve(trimToLimit(request.result as SavedPhoto[]));
    request.onerror = () => resolve([]);
  });
}

export async function savePhoto(dataUrl: string): Promise<void> {
  const db = await openDb();
  if (!db) return;
  const existing = await listSavedPhotos();
  const next = trimToLimit([
    { id: crypto.randomUUID(), dataUrl, savedAt: Date.now() },
    ...existing,
  ]);
  const tx = db.transaction(STORE, 'readwrite');
  const store = tx.objectStore(STORE);
  store.clear();
  for (const photo of next) store.put(photo);
}

export async function forgetPhoto(id: string): Promise<void> {
  const db = await openDb();
  if (!db) return;
  db.transaction(STORE, 'readwrite').objectStore(STORE).delete(id);
}
```

- [ ] **Step 2: Note the interface you are wiring into**

`PhotoUpload` takes `onFileSelected(file: File)`, `onFileClear`, `previewUrl`
and `disabled`, and gets its dictionary from `useTryOnLocale()`. It works in
`File` objects, **not** data URLs, so a cached photo has to be turned back
into a `File` before the existing pipeline will accept it. Add this helper to
`lib/try-on/photo-cache.ts`:

```ts
/* The upload pipeline works in File objects. A cached photo is a data URL,
   so rebuild a File rather than widening every downstream signature. */
export async function savedPhotoToFile(photo: SavedPhoto): Promise<File | null> {
  try {
    const blob = await (await fetch(photo.dataUrl)).blob();
    return new File([blob], `saved-${photo.id}.png`, { type: blob.type || 'image/png' });
  } catch {
    return null;
  }
}
```

Add its props too:

```ts
  allowSavedPhotos?: boolean;
```

- [ ] **Step 4: Save on successful selection**

Inside `handleFile`, after both validation guards pass and before the file is
handed to `onFileSelected`:

```tsx
if (allowSavedPhotos) {
  const reader = new FileReader();
  reader.onload = () => {
    if (typeof reader.result === 'string') void savePhoto(reader.result);
  };
  reader.readAsDataURL(file);
}
```

Saving is fire-and-forget on purpose: a cache write must never delay or block
a generation the shopper is waiting for.
